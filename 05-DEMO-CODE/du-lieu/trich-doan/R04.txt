const $ = (selector, root = document) => root.querySelector(selector);
const $$ = (selector, root = document) => [...root.querySelectorAll(selector)];
// Skip identical catalog requests; changed searches still use native GET forms.
(() => {
  $$('form.catalog-filters[method="get"]').forEach(form => {
    const fields = $$('input[name="q"], select[name="category"], select[name="sort"]', form);
    if (!fields.length) return;
    const state = defaults => JSON.stringify(fields.map(field => {
      const initial = field.tagName === 'SELECT'
        ? ([...field.options].find(option => option.defaultSelected) || field.options[0])?.value || ''
        : field.defaultValue;
      const value = defaults ? initial : field.value;
      return [field.name, field.name === 'q' ? value.trim() : value];
    }));
    const applied = state(true);
    // A legacy courses URL can contain q although its filter form has no q field.
    // Submitting that form clears q, so it must remain a meaningful request.
    const clearsLegacyQuery = !fields.some(field => field.name === 'q')
      && !!new URLSearchParams(location.search).get('q')?.trim();
    form.addEventListener('submit', event => {
      if (!clearsLegacyQuery && state(false) === applied) {
        event.preventDefault();
        document.getElementById('catalog-results')?.scrollIntoView({block: 'start'});
      }
    });
  });
})();
let toastTimer;
function bindImageFallback(image) {
  const fallback = image.dataset.fallback || 'assets/image-fallback.svg';
  const replace = () => {
    if (image.dataset.fallbackApplied) return;
    image.dataset.fallbackApplied = 'true';
    image.src = fallback;
  };
  image.addEventListener('error', replace, {once: true});
  if (image.complete && image.naturalWidth === 0 && image.getAttribute('src')) replace();
}
$$('img[data-fallback]').forEach(bindImageFallback);
function toast(message) {
  const box = $('.toast');
  if (!box) return;
  box.textContent = message;
  box.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => box.classList.remove('show'), 4000);
}
async function postApi(body) {
  const multipart = body instanceof FormData;
  try {
    const response = await fetch('api.php', {
      method: 'POST',
      credentials: 'same-origin',
      headers: { 'X-CSRF-Token': $('meta[name="csrf-token"]').content, ...(multipart ? {} : {'Content-Type': 'application/json'}) },
      body: multipart ? body : JSON.stringify(body),
    });
    const result = await response.json();
    if (result.csrf) $('meta[name="csrf-token"]').content = result.csrf;
    return result;
  } catch {
    return {ok: false, message: 'Không kết nối được máy chủ. Dữ liệu chưa được lưu, hãy thử lại.'};
  }
}

function syncPasswordMatch(showFeedback = false) {
  if (!authForm) return;
  const confirmation = authForm.elements.confirm_password;
  const mismatch = authMode === 'register' && confirmation.value !== '' && confirmation.value !== authForm.elements.password.value;
  confirmation.setCustomValidity(mismatch ? 'Mật khẩu nhập lại chưa khớp.' : '');
  const feedback = $('#password-match');
  if (!mismatch) {
    confirmation.removeAttribute('aria-invalid');
    feedback.textContent = '';
  } else if (showFeedback || confirmation.getAttribute('aria-invalid') === 'true') {
    confirmation.setAttribute('aria-invalid', 'true');
    feedback.textContent = 'Mật khẩu nhập lại chưa khớp.';
  }
}
authForm?.elements.confirm_password.addEventListener('blur', () => syncPasswordMatch(true));

function syncMenu(open) {
  menuTriggers.forEach(button => {
    button.setAttribute('aria-expanded', String(open));
    button.setAttribute('aria-label', button.classList.contains('hero-logo-trigger') ? (open ? 'Đóng danh mục Artio' : 'Mở danh mục Artio') : (open ? 'Đóng menu' : 'Mở menu'));
  });
}
function closeMenu(restoreFocus = false) {
  if (!menu?.open) return;
  const focusInside = menu.contains(document.activeElement);
  menu.close();
  syncMenu(false);
  if (restoreFocus && focusInside) lastMenuTrigger?.focus();
}
