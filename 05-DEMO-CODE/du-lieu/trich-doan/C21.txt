function artio_search_normalize(string $value): string
{
    // Keep search behavior consistent even when mbstring/intl are unavailable.
    $value = strtr($value, [
        'Á'=>'á','À'=>'à','Ả'=>'ả','Ã'=>'ã','Ạ'=>'ạ','Ă'=>'ă','Ắ'=>'ắ','Ằ'=>'ằ','Ẳ'=>'ẳ','Ẵ'=>'ẵ','Ặ'=>'ặ','Â'=>'â','Ấ'=>'ấ','Ầ'=>'ầ','Ẩ'=>'ẩ','Ẫ'=>'ẫ','Ậ'=>'ậ',
        'É'=>'é','È'=>'è','Ẻ'=>'ẻ','Ẽ'=>'ẽ','Ẹ'=>'ẹ','Ê'=>'ê','Ế'=>'ế','Ề'=>'ề','Ể'=>'ể','Ễ'=>'ễ','Ệ'=>'ệ',
        'Í'=>'í','Ì'=>'ì','Ỉ'=>'ỉ','Ĩ'=>'ĩ','Ị'=>'ị','Ó'=>'ó','Ò'=>'ò','Ỏ'=>'ỏ','Õ'=>'õ','Ọ'=>'ọ','Ô'=>'ô','Ố'=>'ố','Ồ'=>'ồ','Ổ'=>'ổ','Ỗ'=>'ỗ','Ộ'=>'ộ','Ơ'=>'ơ','Ớ'=>'ớ','Ờ'=>'ờ','Ở'=>'ở','Ỡ'=>'ỡ','Ợ'=>'ợ',
        'Ú'=>'ú','Ù'=>'ù','Ủ'=>'ủ','Ũ'=>'ũ','Ụ'=>'ụ','Ư'=>'ư','Ứ'=>'ứ','Ừ'=>'ừ','Ử'=>'ử','Ữ'=>'ữ','Ự'=>'ự','Ý'=>'ý','Ỳ'=>'ỳ','Ỷ'=>'ỷ','Ỹ'=>'ỹ','Ỵ'=>'ỵ','Đ'=>'đ',
    ]);
    if (function_exists('mb_strtolower')) $value = mb_strtolower($value, 'UTF-8');
    else $value = strtolower($value);
    if (class_exists('Normalizer')) $value = Normalizer::normalize($value, Normalizer::FORM_D) ?: $value;
    $value = preg_replace('/\p{M}+/u', '', $value) ?: $value;
    return strtr($value, [
        'á'=>'a','à'=>'a','ả'=>'a','ã'=>'a','ạ'=>'a','ă'=>'a','ắ'=>'a','ằ'=>'a','ẳ'=>'a','ẵ'=>'a','ặ'=>'a','â'=>'a','ấ'=>'a','ầ'=>'a','ẩ'=>'a','ẫ'=>'a','ậ'=>'a',
        'é'=>'e','è'=>'e','ẻ'=>'e','ẽ'=>'e','ẹ'=>'e','ê'=>'e','ế'=>'e','ề'=>'e','ể'=>'e','ễ'=>'e','ệ'=>'e','í'=>'i','ì'=>'i','ỉ'=>'i','ĩ'=>'i','ị'=>'i',
        'ó'=>'o','ò'=>'o','ỏ'=>'o','õ'=>'o','ọ'=>'o','ô'=>'o','ố'=>'o','ồ'=>'o','ổ'=>'o','ỗ'=>'o','ộ'=>'o','ơ'=>'o','ớ'=>'o','ờ'=>'o','ở'=>'o','ỡ'=>'o','ợ'=>'o',
        'ú'=>'u','ù'=>'u','ủ'=>'u','ũ'=>'u','ụ'=>'u','ư'=>'u','ứ'=>'u','ừ'=>'u','ử'=>'u','ữ'=>'u','ự'=>'u','ý'=>'y','ỳ'=>'y','ỷ'=>'y','ỹ'=>'y','ỵ'=>'y','đ'=>'d',
    ]);
