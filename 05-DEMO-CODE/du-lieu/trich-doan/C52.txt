async function postApi(body) {
  const multipart = body instanceof FormData;
  try {
    const response = await fetch('api.php', {
      method: 'POST',
      credentials: 'same-origin',
      headers: { 'X-CSRF-Token': $('meta[name="csrf-token"]').content, ...(multipart ? {} : {'Content-Type': 'application/json'}) },
      body: multipart ? body : JSON.stringify(body),
    });
    const result = await response.json();
    if (result.csrf) $('meta[name="csrf-token"]').content = result.csrf;
    return result;
  } catch {
    return {ok: false, message: 'Không kết nối được máy chủ. Dữ liệu chưa được lưu, hãy thử lại.'};
  }
}
