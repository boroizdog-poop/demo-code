function artio_admin_section_image(string $section, array $content, array $courses): string {
    if($section==='beginners') {
        foreach($courses as $course)if($course['status']==='published'&&$course['level']==='Tân thủ')return $course['image_url'];
    }
    if($section==='new_share') {
        $items=[];
        foreach($courses as $course)if($course['status']==='published')$items[]=['date'=>$course['updated_at']??$course['created_at']??'','image'=>$course['image_url']];
        foreach(artio_brushes() as $brush)$items[]=['date'=>$brush['shared_at']??'','image'=>$brush['image']];
        usort($items,fn($a,$b)=>strcmp($b['date'],$a['date']));
        if($items)return $items[0]['image'];
    }
    return $content[artio_admin_sections()[$section]['image']];
}
