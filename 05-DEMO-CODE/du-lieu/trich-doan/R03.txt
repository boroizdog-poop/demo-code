function trackAdminForm(form) {
  formRevisions.set(form,0);
  const changed=()=>{formRevisions.set(form,formRevisions.get(form)+1);dirtyForms.add(form);};
  form.addEventListener('input',changed);
  form.addEventListener('change',changed);
}
function setAdminBusy(form, busy, activeButton) {
  if(busy) form.setAttribute('aria-busy','true'); else form.removeAttribute('aria-busy');
  const buttons=$$('button[type="submit"],button:not([type])',form);
  const active=activeButton||buttons[0];
  buttons.forEach(button=>{
    button.setAttribute('aria-disabled',String(busy));
    if(busy&&button===active) button.setAttribute('aria-busy','true'); else button.removeAttribute('aria-busy');
  });
}

function filterAdminCourses() {
  if(!adminSearch) return;
  const normalize = text => text.normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/đ/gi,'d').toLowerCase();
  const query = normalize(adminSearch.value.trim());
  const rows = $$('[data-admin-title]');
  rows.forEach(row => {
    row.hidden = !normalize(row.dataset.adminTitle).includes(query);
    if(row.dataset.adminBrushId){
      const details=document.getElementById('brush-'+row.dataset.adminBrushId);
      if(details)details.hidden=row.hidden;
    }
  });
  const noResults = $('[data-admin-no-results]');
  if(noResults) noResults.hidden = !query || rows.some(row => !row.hidden);
  const clear = $('.admin-search-clear');
  if(clear) clear.hidden = !adminSearch.value;
}

function savedFormId(form, name, value) {
  let input=form.elements[name];
  if(!input){input=document.createElement('input');input.type='hidden';input.name=name;form.append(input);}
  input.value=String(value);

async function savedAdminDocument(view) {
  const response=await fetch('admin.php?view='+view,{credentials:'same-origin'});
  if(!response.ok)throw new Error('Cannot refresh saved record');
  const page=new DOMParser().parseFromString(await response.text(),'text/html');
  if(!page.querySelector('.admin-content'))throw new Error('Admin session expired');
  const csrf=page.querySelector('meta[name="csrf-token"]');if(csrf)$('meta[name="csrf-token"]').content=csrf.content;
  return page;
}
async function refreshSavedCategory(form,id,revision) {
  const page=await savedAdminDocument('categories');
  if(revision!==formRevisions.get(form))return;
  const saved=[...page.querySelectorAll('.categories-admin .category-form')].find(item=>Number(item.elements.id.value)===Number(id));
  if(!saved)throw new Error('Missing saved category');
  const list=$('.categories-admin');
  const existing=$$('.category-form',list).find(item=>Number(item.elements.id.value)===Number(id));
  if(!existing){list.append(saved);initAdminResourceForms(list);}
  else if(!dirtyForms.has(existing)){
    existing.elements.name.value=saved.elements.name.value;existing.elements.slug.value=saved.elements.slug.value;
  }
  form.reset();form.elements.id?.remove();dirtyForms.delete(form);
}
async function refreshSavedBrush(form,id,revision) {
  const page=await savedAdminDocument('brushes');
  const details=page.getElementById('brush-'+id), list=$('.admin-content .admin-course-list');
  if(!details||!details.previousElementSibling||!list)throw new Error('Missing saved brush');
  const row=details.previousElementSibling;
  const existing=document.getElementById('brush-'+id);
  const isCreation=Boolean(form.closest('#add-brush'));
  const records=[...page.querySelectorAll('details[id^="brush-"]')];
  const following=records.slice(records.indexOf(details)+1).map(item=>document.getElementById(item.id)).find(Boolean);
  const before=following?.previousElementSibling||null;
  if(existing){
    existing.previousElementSibling.replaceWith(row);
    const summary=$('summary',existing), savedSummary=details.querySelector('summary');
    if(summary&&savedSummary)summary.textContent=savedSummary.textContent;
    if(existing.nextElementSibling!==before){
      const focused=document.activeElement;
      const restoreFocus=focused&&existing.contains(focused);
      list.insertBefore(row,before);list.insertBefore(existing,before);
      if(restoreFocus)focused.focus({preventScroll:true});
    }
  } else {
    if(revision!==formRevisions.get(form))return;
    list.insertBefore(row,before);list.insertBefore(details,before);
    initAdminResourceForms(details);initAdminSelects(details);
  }
  if(isCreation&&revision===formRevisions.get(form)){
    form.reset();form.elements.id?.remove();form.elements.version?.remove();
    const preview=$('[data-brush-image-preview]',form), placeholder=$('[data-brush-image-placeholder]',form);
    if(preview){preview.hidden=true;preview.removeAttribute('src');}if(placeholder)placeholder.hidden=false;
    const uploadStatus=$('[data-brush-upload-status]',form);if(uploadStatus)uploadStatus.textContent='';
    dirtyForms.delete(form);
  }
  $$('[data-brush-stat]').forEach(stat=>{
    const saved=page.querySelector('[data-brush-stat="'+stat.dataset.brushStat+'"]');
    if(saved)stat.textContent=saved.textContent;
  });
  filterAdminCourses();
}

    list.addEventListener('keydown',event=>{
      const current=options.indexOf(document.activeElement);
      if(event.key==='ArrowDown'||event.key==='ArrowUp'){event.preventDefault();focusOption((current+(event.key==='ArrowDown'?1:-1)+options.length)%options.length);}
      else if(event.key==='Home'){event.preventDefault();focusOption(0);}
      else if(event.key==='End'){event.preventDefault();focusOption(options.length-1);}
      else if(event.key==='Escape'||event.key==='Tab'){list.hidePopover();button.focus();if(event.key==='Escape')event.preventDefault();}
      else if(event.key.length===1&&event.key!==' '){
        typeahead+=event.key.toLocaleLowerCase('vi');clearTimeout(typeaheadTimer);typeaheadTimer=setTimeout(()=>typeahead='',600);
        const next=options.findIndex(item=>item.textContent.toLocaleLowerCase('vi').startsWith(typeahead));focusOption(next);
      }
    });
    list.addEventListener('toggle',event=>button.setAttribute('aria-expanded',String(event.newState==='open')));
    select.addEventListener('change',sync);
    function responsive() {list.hidePopover();select.hidden=desktopSelect.matches;button.hidden=!desktopSelect.matches;sync();}
    desktopSelect.addEventListener('change',responsive);responsive();
    window.addEventListener('resize',()=>list.hidePopover());
    window.addEventListener('scroll',()=>{if(list.matches(':popover-open'))list.hidePopover();},{passive:true});
  });
}
