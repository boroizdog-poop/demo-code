function artio_csrf(): string { return $_SESSION['csrf'] ??= bin2hex(random_bytes(32)); }
function artio_check_csrf(): void
{
    $token = $_SERVER['HTTP_X_CSRF_TOKEN'] ?? $_POST['csrf'] ?? '';
    if (!is_string($token) || !hash_equals(artio_csrf(), $token)) throw new RuntimeException('Phiên làm việc đã hết hạn. Hãy tải lại trang.', 419);
}
function artio_user(): ?array
{
    // Reuse the lookup for the header, cards and permission checks in this request.
    static $cachedId = 0, $user = null;
    $id = (int)($_SESSION['user_id'] ?? 0);
    if (!$id) { $cachedId = 0; $user = null; return null; }
    if ($cachedId === $id) return $user;
    $stmt = artio_db()->prepare('SELECT id,name,email,username,role FROM users WHERE id=?');
    $stmt->execute([$id]);
    $cachedId = $id;
    return $user = $stmt->fetch() ?: null;
}
function artio_is_admin(): bool { return (artio_user()['role'] ?? '') === 'admin'; }
function artio_require_admin(): void
{
    if (!artio_is_admin()) throw new RuntimeException('Bạn không có quyền quản trị.', 403);
}
