<?php
$path = rawurldecode(parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH) ?? '/');
$public = ['/', '/index.php', '/course.php', '/courses.php', '/search.php', '/brush.php', '/brush-detail.php', '/contact.php', '/admin.php', '/api.php', '/account.php'];
if (in_array($path, $public, true)) { require __DIR__ . ($path === '/' ? '/index.php' : $path); return true; }
$file = realpath(__DIR__ . $path);
foreach (['assets', 'uploads'] as $directory) {
    $root = realpath(__DIR__ . '/' . $directory);
    if ($root && $file && str_starts_with($file, $root . DIRECTORY_SEPARATOR) && is_file($file) && preg_match('/\.(css|js|ttf|woff2|woff|png|jpg|jpeg|webp|svg)$/i', $file)) return false;
}
http_response_code(404); echo 'Không tìm thấy trang.';
