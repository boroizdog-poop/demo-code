'use strict';
const motionPreference = matchMedia('(prefers-reduced-motion: reduce)');
if ('IntersectionObserver' in window) {
  const targets = [...document.querySelectorAll('[data-gallery-reveal], .catalog-body .maku-course-card, .brush-card')];
  const reveal = element => {
    element.classList.remove('is-waiting');
    observer.unobserve(element);
  };
  const observer = new IntersectionObserver(entries => {
    entries.forEach(entry => { if (entry.isIntersecting) reveal(entry.target); });
  }, {threshold: 0.08, rootMargin: '0px 0px -48px 0px'});
  const anchor = document.getElementById(location.hash.slice(1));
  // Each entrance runs once. Visible, focused and directly linked content stays readable.
  targets.forEach(element => {
    const rect = element.getBoundingClientRect();
    if (motionPreference.matches || rect.top < innerHeight || element.contains(document.activeElement) || (anchor && (anchor === element || anchor.contains(element)))) return;
    element.classList.add('gallery-reveal', 'is-waiting');
    observer.observe(element);
  });
  const revealAll = () => targets.forEach(reveal);
  // Tab or other focus navigation reveals only the focused content and its ancestors.
  document.addEventListener('focusin', event => {
    let target = event.target.closest('.gallery-reveal');
    while (target) {
      reveal(target);
      target = target.parentElement?.closest('.gallery-reveal');
    }
  });
  motionPreference.addEventListener('change', event => {
    if (event.matches) { revealAll(); observer.disconnect(); }
  });
  window.addEventListener('pageshow', event => { if (event.persisted) revealAll(); });
}
const carousel = document.querySelector('[data-carousel]');
if (carousel) {
  const slides = [...carousel.querySelectorAll('[data-slide]')];
  const previous = document.querySelector('[data-carousel-prev]');
  const next = document.querySelector('[data-carousel-next]');
  const status = document.querySelector('[data-carousel-status]');
  const toggle = document.querySelector('[data-carousel-toggle]');
  const pauseReasons = new Set();
  let active = 0, pendingFrame = 0, autoTimer = 0, autoDirection = 1;
  // Advance only while visible; reader interaction pauses autoplay.
  let userPaused = false, resumeTimer = 0;
  pauseReasons.add('offscreen');
  const select = index => {
    active = index;
    slides.forEach((slide, i) => slide.classList.toggle('is-active', i === active));
    previous.disabled = active === 0;
    next.disabled = active === slides.length - 1;
    status.textContent = `${String(active + 1).padStart(2, '0')} / ${String(slides.length).padStart(2, '0')}`;
  };
  const go = (index, keyboard = false, automatic = false) => {
    if (!slides.length) return;
    index = Math.max(0, Math.min(slides.length - 1, index));
    if (!automatic) pauseBriefly();
    select(index);
    const slide = slides[index];
    carousel.scrollTo({left: slide.offsetLeft - (carousel.clientWidth - slide.offsetWidth) / 2,
      behavior: keyboard || motionPreference.matches ? 'instant' : 'smooth'});
  };
  previous.addEventListener('click', event => go(active - 1, event.detail === 0));
  next.addEventListener('click', event => go(active + 1, event.detail === 0));
  carousel.addEventListener('keydown', event => {
    if (!['ArrowLeft', 'ArrowRight', 'Home', 'End'].includes(event.key)) return;
    event.preventDefault();
    const index = event.key === 'Home' ? 0 : event.key === 'End' ? slides.length - 1 : active + (event.key === 'ArrowRight' ? 1 : -1);
    go(index, true);
    slides[active]?.querySelector('a').focus({preventScroll: true});
  });
  carousel.addEventListener('focusin', event => {
    const index = slides.findIndex(slide => slide.contains(event.target));
    if (index >= 0 && index !== active) go(index, true);
  });
  carousel.addEventListener('scroll', () => {
    if (pendingFrame) return;
    pendingFrame = requestAnimationFrame(() => {
      pendingFrame = 0;
      const center = carousel.scrollLeft + carousel.clientWidth / 2;
      let closest = 0, distance = Infinity;
      slides.forEach((slide, index) => {
        const delta = Math.abs(slide.offsetLeft + slide.offsetWidth / 2 - center);
        if (delta < distance) { distance = delta; closest = index; }
      });
      if (closest !== active) select(closest);
    });
  }, {passive: true});
  window.addEventListener('resize', () => {
    if (!slides.length) return;
    carousel.scrollTo({left: slides[active].offsetLeft - (carousel.clientWidth - slides[active].offsetWidth) / 2, behavior: 'instant'});
  }, {passive: true});
  if (slides.length) select(0);
  else { previous.disabled = true; next.disabled = true; status.textContent = 'Chưa có tài nguyên'; }

  // Native smooth scrolling between anchors avoids a perpetual scroll loop.
  // Focus, direct interaction and reduced motion keep control with the reader.
  function syncPlayback() {
    const available = !motionPreference.matches && slides.length > 1;

    toggle.hidden = !available;
    const playing = available && !document.hidden && !userPaused && pauseReasons.size === 0;
    status.setAttribute('aria-live', playing ? 'off' : 'polite');
    toggle.setAttribute('aria-pressed', String(userPaused));
    toggle.setAttribute('aria-label', userPaused ? 'Tiếp tục chuyển động' : 'Tạm dừng chuyển động');
    toggle.firstElementChild.textContent = userPaused ? '▷' : 'Ⅱ';
    if (!playing) {
      clearTimeout(autoTimer);
      autoTimer = 0;
    } else if (!autoTimer) {
      autoTimer = setTimeout(tick, 4200);
    }
  }
  function pause(reason, paused) {
    paused ? pauseReasons.add(reason) : pauseReasons.delete(reason);
    syncPlayback();
  }
  function pauseBriefly() {
    clearTimeout(resumeTimer);
    pause('interaction', true);
    resumeTimer = setTimeout(() => pause('interaction', false), 3500);
  }
  function tick() {
    autoTimer = 0;
    if (active === slides.length - 1) autoDirection = -1;
    if (active === 0) autoDirection = 1;
    go(active + autoDirection, false, true);
    syncPlayback();
  }
  if ('IntersectionObserver' in window) {
    const visibility = new IntersectionObserver(entries => pause('offscreen', !entries[0].isIntersecting), {threshold: 0.1});
    visibility.observe(carousel);
  } else pauseReasons.delete('offscreen');
  carousel.addEventListener('focusin', () => pause('focus', true));
  carousel.addEventListener('focusout', event => { if (!carousel.contains(event.relatedTarget)) pause('focus', false); });
  carousel.addEventListener('pointerdown', () => pause('pointer', true), {passive: true});
  window.addEventListener('pointerup', () => { if (pauseReasons.has('pointer')) { pauseBriefly(); pause('pointer', false); } }, {passive: true});
  window.addEventListener('pointercancel', () => pause('pointer', false), {passive: true});
  carousel.addEventListener('wheel', pauseBriefly, {passive: true});
  carousel.addEventListener('pointerenter', event => { if (event.pointerType === 'mouse') pause('hover', true); });
  carousel.addEventListener('pointerleave', () => pause('hover', false));
  toggle.addEventListener('click', () => { userPaused = !userPaused; syncPlayback(); });
  document.addEventListener('visibilitychange', syncPlayback);
  motionPreference.addEventListener('change', syncPlayback);
  syncPlayback();
}
